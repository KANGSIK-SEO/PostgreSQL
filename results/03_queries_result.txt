-- [집계 2 - AVG] 출처(source)별 작품 평균 제작연도 집계
SELECT source, ROUND(AVG(year_start), 1) AS avg_year_start
FROM artworks
GROUP BY source
ORDER BY avg_year_start;

) AND is_premium = FALSE;
